<script>

  // Replace this with your Gemini API key.
  // For production applications, NEVER expose your API key
  // directly in frontend JavaScript.
  const GEMINI_API_KEY = "YOUR_GEMINI_API_KEY";

  async function generatePlan() {

    const age = document.getElementById("age").value;
    const level = document.getElementById("level").value;
    const goal = document.getElementById("goal").value;
    const days = document.getElementById("days").value;
    const duration = document.getElementById("duration").value;
    const equipment = document.getElementById("equipment").value;

    if (!age || !days || !duration) {
      alert("Please fill in all required fields.");
      return;
    }

    const result = document.getElementById("result");

    result.innerHTML =
      "<span class='loading'>Generating your personalized plan...</span>";

    const prompt = `
You are FitBuddy-AI, a helpful fitness planning assistant.

Create a personalized weekly workout plan using these details:

Age: ${age}
Fitness Level: ${level}
Fitness Goal: ${goal}
Workout Days Per Week: ${days}
Workout Duration: ${duration} minutes
Available Equipment: ${equipment || "No equipment"}

Provide:

1. Weekly workout schedule
2. Exercises for each day
3. Sets and repetitions
4. Rest periods
5. Warm-up and cool-down
6. Short fitness tips

Keep the plan clear and beginner-friendly where appropriate.
Do not diagnose medical conditions or provide medical treatment.
`;

    try {

      const response = await fetch(
        "https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key="
        + GEMINI_API_KEY,
        {
          method: "POST",

          headers: {
            "Content-Type": "application/json"
          },

          body: JSON.stringify({
            contents: [
              {
                parts: [
                  {
                    text: prompt
                  }
                ]
              }
            ]
          })
        }
      );

      const data = await response.json();

      if (!response.ok) {
        throw new Error(data.error?.message || "API request failed");
      }

      const plan =
        data.candidates?.[0]?.content?.parts?.[0]?.text;

      result.textContent =
        plan || "Unable to generate a fitness plan.";

    } catch (error) {

      console.error(error);

      result.textContent =
        "Sorry, something went wrong while generating your plan.\n\n"
        + error.message;
    }
  }

</script>
